3.2 Canonical Data Transfer Objects (DTOs)
TypeScript
export interface StructuredTermsDTO {
  title: string;
  deliverables: string[];           // Ordered list of concrete deliverables
  priceMinorUnits: number;          // e.g. 98000 for $980.00 USD (NEVER float)
  currency: string;                 // ISO-4217 e.g. "USD"
  deliveryDays: number;             // Calendar days from agreement activation
  includedRevisions: number;        // Integer >= 0
  acceptanceCriteriaText: string;
}
export interface MandateDTO {
  mandateId: string;
  dealId: string;
  partyRole: PartyRole;
  version: number;
  currency: string;
  limitPriceMinorUnits: number;     // Max budget for BUYER; Min floor for SELLER
  targetPriceMinorUnits: number;
  limitDeliveryDays: number;        // Max days for BUYER; Min days for SELLER
  targetDeliveryDays: number;
  limitRevisions: number;           // Min revisions for BUYER; Max revisions for SELLER
  targetRevisions: number;
  mustHaveDeliverables: string[];
  forbiddenTerms: string[];
  mode: NegotiationMode;
  maxRounds: number;
  expiresAt: string;                // ISO-8601 UTC
}
export interface OfferDTO {
  offerId: string;
  dealId: string;
  roundNumber: number;
  proposerRole: PartyRole;
  generatedByAgent: boolean;
  terms: StructuredTermsDTO;
  rationaleSummary: string;         // AI-generated trade-off explanation (max 500 chars)
  callerUtilityScore: string;       // BigDecimal serialized e.g. "0.7850"
  mandateCompliant: boolean;
  createdAt: string;
}
export interface AgreementSnapshotDTO {
  agreementId: string;
  dealId: string;
  version: number;
  canonicalTerms: StructuredTermsDTO;
  termsHashSha256: string;          // SHA-256 of RFC-8785 canonical JSON of canonicalTerms
  buyerAcceptedAt: string | null;
  sellerAcceptedAt: string | null;
}
export interface SettlementStateDTO {
  dealId: string;
  agreementVersion: number;
  invoiceStatus: InvoiceLifecycleStatus;
  paypalInvoiceId: string | null;
  paypalInvoiceNumber: string | null;
  paypalInvoiceViewUrl: string | null;
  paymentStatus: PaymentLifecycleStatus;
  paypalOrderId: string | null;
  paypalAuthorizationId: string | null;
  paypalCaptureId: string | null;
  authorizationExpiresAt: string | null;
  amountMinorUnits: number;
  currency: string;
  lastReconciledAt: string | null;
}
export interface DeliverableEvidenceDTO {
  evidenceId: string;
  dealId: string;
  uploaderRole: PartyRole;
  filename: string;
  mimeType: string;
  sizeBytes: number;
  sha256Hash: string;
  version: number;
  downloadSignedUrl: string;
  uploadedAt: string;
}
export interface DealWorkspaceDTO {
  dealId: string;
  status: DealStatus;
  stopReason: NegotiationStopReason;
  buyerEmail: string;
  sellerEmail: string | null;
  myRole: PartyRole;
  myMandate: MandateDTO | null;
  offers: OfferDTO[];
  currentAgreement: AgreementSnapshotDTO | null;
  settlement: SettlementStateDTO;
  evidenceList: DeliverableEvidenceDTO[];
  disputeStatus: DisputeStatus;
  disputeReason: string | null;
}
